\newbox\BookMissaleTableBox
\newdimen\BookMissaleTableHeight
\newdimen\BookMissaleRowAdjustment
\newdimen\BookMissaleTableRemainder
\newcommand{\BookMissaleTable}[2]{%
  \setbox\BookMissaleTableBox=\hbox{#2}%
  \ifnum#1=0
    \global\BookMissaleTableHeight=\dimexpr\ht\BookMissaleTableBox+\dp\BookMissaleTableBox\relax
  \else
    % Distribute the height difference among the body rows, keeping the
    % type and the common header at their original size.
    \BookMissaleTableRemainder=\dimexpr\BookMissaleTableHeight-\ht\BookMissaleTableBox-\dp\BookMissaleTableBox\relax
    \BookMissaleRowAdjustment=\dimexpr\BookMissaleTableRemainder/#1\relax
    \advance\BookMissaleTableRemainder by\dimexpr-\BookMissaleRowAdjustment*#1\relax
    \setbox\BookMissaleTableBox=\hbox{#2}%
  \fi
  \nointerlineskip\box\BookMissaleTableBox
}
