% Assemble only the supplied details, in their printed order.
\makeatletter
\newcommand*{\reading@details}{}
\newcommand*{\reading@adddetail}[1]{%
  \ifx\reading@details\@empty
    \protected@edef\reading@details{#1}%
  \else
    \protected@edef\reading@details{\reading@details\space\textperiodcentered\space#1}%
  \fi
}
\newcommand{\makereadingtitle}{%
  \begin{center}
    {\small\MakeUppercase{\@title}\par}
    \vspace{2pt}
    {\large\bfseries\thereadingtype\par}
    \let\reading@details\@empty
    \ifx\theyearcycle\@empty\else\reading@adddetail{Anno~\theyearcycle}\fi
    \ifx\themassdesignation\@empty\else\reading@adddetail{\themassdesignation}\fi
    \ifx\thereadingnumber\@empty\else\reading@adddetail{N.~\thereadingnumber}\fi
    \ifx\thetone\@empty\else\reading@adddetail{\thetone}\fi
    \ifx\thereadingform\@empty\else\reading@adddetail{\thereadingform}\fi
    \ifx\thereadingnote\@empty\else\reading@adddetail{\thereadingnote}\fi
    \ifx\reading@details\@empty\else
      \vspace{1.5pt}
      {\small\reading@details\par}
    \fi
    \vspace{2pt}
    {\small\color{gregoriocolor}\thescriptureref\par}
  \end{center}
}
\makeatother
