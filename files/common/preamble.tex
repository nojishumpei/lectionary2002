% Print command
\makeatletter
\newcommand{\makereadingtitle}{%
  \begin{center}
    {\large \@title\par}
    \vspace{1ex}
    {% A reading option number occupies the same line as a year cycle.
      \ifx\thereadingnumber\@empty
        \theyearcycle
      \else
        \thereadingnumber
      \fi\par}
    \vspace{1ex}
    {\color{gregoriocolor} \thescriptureref\par}
    \vspace{1ex}
    {\small \textbf{\thetone} \par}
  \end{center}
}
\makeatother
